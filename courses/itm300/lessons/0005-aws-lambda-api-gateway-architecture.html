3. The Lambda Handler Pattern

// lambda.mjs
import { DateTime } from 'luxon';
import admin from 'firebase-admin';

if (!admin.apps.length) {
  admin.initializeApp();
}

export async function handler(event) {
  const path = event.rawPath || event.path;
  const authHeader = event.headers?.authorization || event.headers?.Authorization;

  // 1. Verify Firebase ID Token
  if (!authHeader?.startsWith('Bearer ')) {
    return { statusCode: 401, body: JSON.stringify({ error: 'Missing token' }) };
  }

  const token = authHeader.slice(7);
  let decodedToken;
  try {
    decodedToken = await admin.auth().verifyIdToken(token);
  } catch (err) {
    return { statusCode: 401, body: JSON.stringify({ error: 'Invalid token' }) };
  }

  // 2. Server-side Role Check from Firestore
  const userDoc = await admin.firestore().collection('users').doc(decodedToken.uid).get();
  if (userDoc.data()?.role !== 'manager') {
    return { statusCode: 403, body: JSON.stringify({ error: 'Manager access required' }) };
  }

  // 3. Route to your existing calculations...
  if (path === '/reports/weekly/preview') {
    const weekStart = event.queryStringParameters?.weekStart;
    // ... same calculations you wrote today!
    return {
      statusCode: 200,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ approvedHours: "1.75", pendingHours: "2.00" })
    };
  }

  return { statusCode: 404, body: JSON.stringify({ error: 'Not Found' }) };
}

4. AWS Cost & Free Tier Analysis

Before deploying any cloud infrastructure, let's look at the actual costs:

  • AWS Lambda Free Tier: 1,000,000 requests per month and 400,000 GB-seconds of compute time per month permanently free. A 128MB function executing in 200ms costs $0.00.
  • API Gateway (HTTP API v2): 1,000,000 requests per month free for the first 12 months, then only $1.00 per million requests.
  • Secrets & Credentials: Storing Firebase service account JSON as an encrypted Lambda environment variable costs $0.00 (AWS Secrets Manager charges $0.40/month per secret).
  • Total Estimated AWS Cost: $0.00 / month during development and testing.
